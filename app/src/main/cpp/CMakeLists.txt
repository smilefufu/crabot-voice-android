cmake_minimum_required(VERSION 3.22.1)
project(crabot_voice)
set(CMAKE_CXX_STANDARD 17)
set(DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../../../../native-deps")
add_library(sherpa_c SHARED IMPORTED)
set_target_properties(sherpa_c PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/../jniLibs/${ANDROID_ABI}/libsherpa-onnx-c-api.so")
add_library(sherpa_cxx SHARED IMPORTED)
set_target_properties(sherpa_cxx PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/../jniLibs/${ANDROID_ABI}/libsherpa-onnx-cxx-api.so")
add_library(crabot_voice SHARED voice.cpp)
target_include_directories(crabot_voice PRIVATE "${DEPS}/include")
target_link_libraries(crabot_voice PRIVATE sherpa_cxx sherpa_c log)
