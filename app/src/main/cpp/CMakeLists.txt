cmake_minimum_required(VERSION 3.22.1)
project(crabot_voice)
set(CMAKE_CXX_STANDARD 17)
set(DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../../../../native-deps")
add_library(sherpa_c SHARED IMPORTED)
set_target_properties(sherpa_c PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/../jniLibs/${ANDROID_ABI}/libsherpa-onnx-c-api.so")
add_library(sherpa_cxx SHARED IMPORTED)
set_target_properties(sherpa_cxx PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/../jniLibs/${ANDROID_ABI}/libsherpa-onnx-cxx-api.so")
set(APM "${CMAKE_CURRENT_SOURCE_DIR}/../../../../native-apm")
if(NOT EXISTS "${APM}/manifest.json")
  message(FATAL_ERROR "Run tools/prepare-apm.py with the pinned Meson/Ninja/NDK first")
endif()
file(GLOB APM_LIBRARIES "${APM}/lib/*.a")
list(LENGTH APM_LIBRARIES APM_LIBRARY_COUNT)
if(NOT APM_LIBRARY_COUNT EQUAL 24)
  message(FATAL_ERROR "Incomplete pinned APM dependency set")
endif()
add_library(crabot_voice SHARED voice.cpp echo_processor.cpp echo_jni.cpp)
target_compile_definitions(crabot_voice PRIVATE WEBRTC_POSIX WEBRTC_ANDROID WEBRTC_LINUX WEBRTC_APM_DEBUG_DUMP=0)
target_include_directories(crabot_voice PRIVATE "${APM}/include")
target_include_directories(crabot_voice PRIVATE "${DEPS}/include")
target_link_libraries(crabot_voice PRIVATE sherpa_cxx sherpa_c "-Wl,--start-group" ${APM_LIBRARIES} "-Wl,--end-group" log)
